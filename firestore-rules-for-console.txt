rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper function to check if user is the Super Admin
    function isSuperAdmin() {
      return request.auth != null && request.auth.token.email == 'hitechsoftware03@gmail.com';
    }

    // Helper to check if user has access to a specific portal
    function hasPortalAccess(portal) {
      let userDoc = get(/databases/$(database)/documents/users/$(request.auth.uid));
      return isSuperAdmin() || (userDoc != null && portal in userDoc.data.accessiblePortals);
    }

    // User profiles: Users can read their own; Super Admin and
    // Communications staff (internal staff directory) can read all.
    match /users/{userId} {
      allow read: if request.auth != null && (request.auth.uid == userId || isSuperAdmin() || hasPortalAccess('communications'));
      allow write: if isSuperAdmin();
    }

    // Managed Content (Web Management)
    match /news/{id} { allow read: if true; allow write: if isSuperAdmin() || hasPortalAccess('web-management'); }
    match /services/{id} { allow read: if true; allow write: if isSuperAdmin() || hasPortalAccess('web-management'); }
    match /team/{id} { allow read: if true; allow write: if isSuperAdmin() || hasPortalAccess('web-management'); }
    match /testimonials/{id} { allow read: if true; allow write: if isSuperAdmin() || hasPortalAccess('web-management'); }
    match /banners/{id} { allow read: if true; allow write: if isSuperAdmin() || hasPortalAccess('web-management'); }

    // Operational Data
    match /attendance/{id} {
      allow read: if request.auth != null && (resource.data.userId == request.auth.uid || isSuperAdmin());
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      // Super Admin can change anything (e.g. approve/reject). A worker may
      // only punch themselves out once — setting punchOutTime and nothing
      // else — so the client can't self-approve or edit its own record.
      allow update: if isSuperAdmin() || (
        request.auth != null &&
        resource.data.userId == request.auth.uid &&
        !('punchOutTime' in resource.data) &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['punchOutTime'])
      );
    }

    match /tasks/{id} {
      allow read: if request.auth != null && (resource.data.assignedTo == request.auth.uid || isSuperAdmin() || hasPortalAccess('system'));
      allow write: if isSuperAdmin() || hasPortalAccess('system');
    }

    match /projectInquiries/{id} {
      allow create: if true;
      allow read, write: if hasPortalAccess('clients');
    }

    match /projects/{id} {
      allow read: if request.auth != null;
      allow write: if isSuperAdmin() || hasPortalAccess('system');
    }

    match /requisitions/{id} {
      allow read: if request.auth != null && (resource.data.userId == request.auth.uid || isSuperAdmin() || hasPortalAccess('system'));
      allow create: if request.auth != null;
      allow update: if isSuperAdmin() || hasPortalAccess('system');
    }

    // Worker advance/retainer requests: a worker can create and read their
    // own; only Super Admin (or system-portal staff) can approve/reject.
    match /retainerRequests/{id} {
      allow read: if request.auth != null && (resource.data.userId == request.auth.uid || isSuperAdmin() || hasPortalAccess('system'));
      allow create: if request.auth != null;
      allow update: if isSuperAdmin() || hasPortalAccess('system');
    }

    // Public contact form submissions (Contact.tsx on the homepage)
    match /contactMessages/{id} {
      allow create: if true;
      allow read, update, delete: if isSuperAdmin() || hasPortalAccess('communications');
    }

    // Newsletter subscribers (SubscribePopup.tsx, site-wide entry popup)
    match /subscribers/{id} {
      allow create: if true;
      allow read, update, delete: if isSuperAdmin() || hasPortalAccess('communications');
    }

    // Sent newsletter/campaign history (admin Communications > Messages)
    match /campaigns/{id} {
      allow read, write: if isSuperAdmin() || hasPortalAccess('communications');
    }

    // File metadata for admin Communications > Files. The actual bytes live
    // on Cloudinary (unsigned upload, no Firebase Storage/Blaze plan
    // needed); this doc just records the resulting URL so the tab can list
    // and delete without calling Cloudinary's (secret-key-gated) Admin API.
    match /files/{id} {
      allow read, write: if isSuperAdmin() || hasPortalAccess('communications');
    }

    // Site-wide editable content (Web Management > Global Config)
    match /siteConfig/{id} {
      allow read: if true;
      allow write: if isSuperAdmin() || hasPortalAccess('web-management');
    }

    // Private integration credentials (e.g. settings/email holds the
    // Resend API key used by /api/contact). Super Admin only, unlike
    // siteConfig above — this must never be publicly readable since it
    // holds a live API key, not display copy.
    match /settings/{id} {
      allow read, write: if isSuperAdmin();
    }

    // Client Ecosystem sales ledger
    match /quotations/{id} {
      allow read, write: if isSuperAdmin() || hasPortalAccess('clients');
    }
    match /invoices/{id} {
      allow read, write: if isSuperAdmin() || hasPortalAccess('clients');
    }
    match /lpos/{id} {
      allow read, write: if isSuperAdmin() || hasPortalAccess('clients');
    }

    match /officeCodes/{id} {
      allow read: if request.auth != null;
      allow write: if isSuperAdmin();
    }

    match /jobApplications/{id} {
      allow create: if true;
      allow read, write: if hasPortalAccess('talent');
    }

    // Marketplace: digital products, service packages & tech gadgets
    match /marketplaceProducts/{id} {
      allow read: if true;
      allow write: if isSuperAdmin() || hasPortalAccess('marketplace');
    }

    // Marketplace orders: anyone can place an order; only the buyer (if
    // signed in) or marketplace staff can read it; only staff can update
    // status while a manual mobile-money/bank payment is verified.
    match /marketplaceOrders/{id} {
      allow create: if true;
      allow read: if isSuperAdmin() || hasPortalAccess('marketplace') || (request.auth != null && resource.data.userId == request.auth.uid);
      allow update, delete: if isSuperAdmin() || hasPortalAccess('marketplace');
    }
  }
}
